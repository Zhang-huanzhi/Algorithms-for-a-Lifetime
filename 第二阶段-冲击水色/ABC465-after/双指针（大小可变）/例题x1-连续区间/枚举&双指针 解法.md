# 枚举
``` python
import sys  
  
def main():  
  
    n, s, l = map(int, sys.stdin.readline().strip().split())  
  
    arr = list(map(int, sys.stdin.readline().strip().split()))  
  
    pre = [0] * n  
    for i in range(n - 1):  
        pre[i + 1] = pre[i] + arr[i]  
  
    # N比较小，可以枚举  
    ans = 1  
    for left in range(1, n + 1):  
        for right in range(left, n + 1):  
  
            if s < left or s > right:  
                continue  
  
            cnt = right - left + 1  
  
            cost_left = pre[s - 1] - pre[left - 1]  
            cost_right = pre[right - 1] - pre[s - 1]  
  
            cost = min(2 * cost_left + cost_right, cost_left + 2 * cost_right)  
  
            if cost <= l:  
                ans = max(ans, cnt)  
  
    print(ans)  
  
  
if __name__ == "__main__":  
    main()
```

# 双指针
``` python
import sys


def main():
    n, s, l = map(int, sys.stdin.readline().strip().split())
    arr = list(map(int, sys.stdin.readline().strip().split()))

    # 保留你习惯的前缀和定义 (大小为 n)
    pre = [0] * n
    for i in range(n - 1):
        pre[i + 1] = pre[i] + arr[i]

    ans = 1
    right = s  # 右指针初始在起点 s

    # 左指针 left 从 1 枚举到 s
    for left in range(1, s + 1):
        # 城镇 left 到城镇 s 的距离
        cost_left = pre[s - 1] - pre[left - 1]

        # 尝试将 right 尽量往右推
        while right + 1 <= n:
            # 试探如果走到 right + 1 的开销
            next_cost_right = pre[(right + 1) - 1] - pre[s - 1]
            cost = min(
                2 * cost_left + next_cost_right, cost_left + 2 * next_cost_right
            )

            # 如果花费 <= l，说明可以到达 right + 1，右指针右移
            if cost <= l:
                right += 1
            else:
                # 超过了预算，右指针推不动了，停止
                break

        # 检查当前的 [left, right] 是否合法
        cost_right = pre[right - 1] - pre[s - 1]
        cost = min(2 * cost_left + cost_right, cost_left + 2 * cost_right)
        if cost <= l:
            ans = max(ans, right - left + 1)

    print(ans)


if __name__ == "__main__":
    main()
```
