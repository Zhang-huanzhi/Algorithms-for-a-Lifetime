``` python
import sys  
from collections import Counter  
  
def main():  
  
    n = int(sys.stdin.readline().strip())  
  
    arr = list(map(int, sys.stdin.readline().strip().split()))  
  
    memo = Counter()  
    memo[0] = 1  
  
    # 异或前缀和  
    pre = [0] * (n + 1)  
  
    for i in range(1, n + 1):  
        pre[i] = pre[i - 1] ^ arr[i - 1]  
  
    ans = 0  
    for i in range(1, n + 1):  
  
        ans += memo[pre[i]]  
  
        memo[pre[i]] += 1  
  
    print(ans)  
  
if __name__ == "__main__":  
    main()
```
