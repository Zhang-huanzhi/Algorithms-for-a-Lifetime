### 并查集
``` python
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.size = [1] * n  # 优化：维护连通块大小
        self.part = n        # 优化：维护连通分量数量

    def find(self, x):
        # 优化：迭代路径减半（防 Python 爆栈，速度极快）
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, x, y):
        root_x = self.find(x)
        root_y = self.find(y)
        
        # 优化：判断是否已连通，常用于判环
        if root_x == root_y:
            return False
            
        # 优化：按大小合并 (小树挂大树)
        if self.size[root_x] > self.size[root_y]:
            root_x, root_y = root_y, root_x
            
        self.parent[root_x] = root_y
        self.size[root_y] += self.size[root_x]
        self.part -= 1  # 成功合并，总连通块减 1
        
        return True

    def get_size(self, x):
        return self.size[self.find(x)]
```

### 逆元
``` python
MOD = 10**9 + 7

def mod_inv(a, mod=MOD):
    """计算 a 的逆元"""
    return pow(a, mod - 2, mod)

# 使用示例
inv_3 = mod_inv(3)
print(inv_3)  # 3 的逆元

# 验证
print((3 * inv_3) % MOD)  # 输出：1
```

### 欧拉筛
``` python
	def linear_sieve(n):  
    is_prime = [True] * (n + 1)  
    primes = []  
    is_prime[0] = is_prime[1] = False  
  
    for i in range(2,n + 1):  
        if is_prime[i]:  
            primes.append(i)  
        for p in primes:  
            if i * p > n:  
                break  
            is_prime[i * p] = False  
            if i % p == 0:  
                break  
    return primes
```
