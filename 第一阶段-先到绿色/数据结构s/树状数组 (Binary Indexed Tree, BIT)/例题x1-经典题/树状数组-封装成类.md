class BIT:  
    def __init__(self, n):  
        self.n = n  
        self.tree = [0] * (n + 1)  
  
    def lowbit(self, x):  
        return x & -x  
  
    def add(self, i, delta):  
        while i <= self.n:  
            self.tree[i] += delta  
            i += self.lowbit(i)  
  
    def query(self, i):  
        res = 0  
        while i > 0:  
            res += self.tree[i]  
            i -= self.lowbit(i)  
        return  res
