## 版本一：类封装版（推荐，结构清晰）

适合理解树状数组的完整结构，代码可读性强。

```python
class BIT:
    def __init__(self, n):
        """初始化树状数组
        
        Args:
            n: 数据的个数
        """
        self.n = n
        self.tree = [0] * (n + 1)  # 下标从 1 开始
    
    def lowbit(self, x):
        """核心操作：提取二进制最低位的 1
        
        Args:
            x: 当前下标
            
        Returns:
            x 的二进制表示中最低位 1 对应的数值
        """
        return x & -x
    
    def update(self, i, delta):
        """单点更新：在第 i 个位置增加 delta
        
        Args:
            i: 更新位置（从 1 开始）
            delta: 增加的值
        """
        while i <= self.n:
            self.tree[i] += delta
            i += self.lowbit(i)  # 向上更新父节点
    
    def query(self, i):
        """前缀和查询：返回 [1, i] 的和
        
        Args:
            i: 查询位置（从 1 开始）
            
        Returns:
            前缀和
        """
        res = 0
        while i > 0:
            res += self.tree[i]
            i -= self.lowbit(i)  # 向下累加子区间
        return res
    
    def query_range(self, l, r):
        """区间和查询：返回 [l, r] 的和
        
        Args:
            l: 左边界
            r: 右边界
            
        Returns:
            区间和
        """
        return self.query(r) - self.query(l - 1)


# === 使用示例 ===
bit = BIT(10)           # 创建长度为 10 的树状数组
bit.update(3, 5)        # 在下标 3 的位置加 5
print(bit.query(3))     # 查询前 3 个数的和
print(bit.query_range(2, 5))  # 查询 [2, 5] 的区间和
```
