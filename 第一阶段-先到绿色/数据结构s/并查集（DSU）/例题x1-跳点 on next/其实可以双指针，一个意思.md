``` python
import sys

# 提高 Python 的递归深度并优化输入输出速度
sys.setrecursionlimit(2000000)

def solve():
    input = sys.stdin.read
    data = input().split()
    
    if not data:
        return
    
    N = int(data[0])
    Q = int(data[1])
    
    # nxt[i] 表示版本 i 当前对应的实际最高版本（并查集思想）
    nxt = list(range(N + 2))
    # count[i] 表示当前版本为 i 的 PC 数量
    count = [1] * (N + 1)
    
    # 并查集的路径压缩查找
    def find(i):
        path = []
        while nxt[i] != i:
            path.append(i)
            i = nxt[i]
        for node in path:
            nxt[node] = i
        return i

    idx = 2
    out = []
    
    for _ in range(Q):
        X = int(data[idx])
        Y = int(data[idx+1])
        idx += 2
        
        upgraded_count = 0
        curr = 1
        
        while True:
            # 找到当前 >= curr 且真实存在的最小版本
            curr = find(curr)
            
            # 如果这个版本已经大于 X，说明所有 <= X 的版本都已经处理完毕
            if curr > X:
                break
                
            # 累加当前版本的 PC 数量
            upgraded_count += count[curr]
            count[curr] = 0 # 它们都被升级了，当前版本的数量清空
            
            # 将当前版本合并到 Y
            nxt[curr] = Y
            
            # 继续检查下一个版本
            curr += 1
            
        # 将所有升级的 PC 数量加到新版本 Y 上
        count[Y] += upgraded_count
        out.append(str(upgraded_count))
        
    print('\n'.join(out))

if __name__ == '__main__':
    solve()
```
