对于冲击 AtCoder 绿牌（Rating 800~1200）来说，掌握这三个核心方法（Init, Find, Union）已经覆盖了 95% 的并查集题目！

``` python
class UnionFind:  
    def __init__(self, n):  
        self.parent = list(range(n+1))  
  
    def find(self, x):  
        if self.parent[x] == x:  
            return x  
        self.parent[x] = self.find(self.parent[x])  
        return self.parent[x]  
  
    def union(self, x, y):  
        root_x = self.find(x)  
        root_y = self.find(y)  
        if root_x != root_y:  
            self.parent[root_x] = root_y
```

``` python
import sys
# 【关键】Python 默认递归深度只有 1000，做并查集/图论题必加这一句！
sys.setrecursionlimit(200000) 

class UnionFind:
    def __init__(self, n):
        # 初始化：生成一个列表 [0, 1, 2, ..., n-1]
        # self.parent[i] 的值就是元素 i 的父节点
        self.parent = list(range(n))
        
        # 可选：记录每个集合的大小，用于“按秩合并”或查询连通块大小
        self.size = [1] * n 

    def find(self, x):
        # 查找掌门，并进行【路径压缩】
        if self.parent[x] == x:
            return x
        # 递归找老大的老大，顺便把当前节点直接连到老大下面
        self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        # 合并两个集合
        root_x = self.find(x)
        root_y = self.find(y)
        
        if root_x != root_y:
            # 把 x 的掌门挂到 y 的掌门下面
            self.parent[root_x] = root_y
            self.size[root_y] += self.size[root_x] # 更新集合大小

    def same(self, x, y):
        # 判断是否同源
        return self.find(x) == self.find(y)
```
