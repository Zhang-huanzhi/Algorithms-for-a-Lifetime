class ListNode:  
    """链表节点定义"""  
  
    def __init__(self, val=0, next=None):  
        self.val = val  
        self.next = next  
  
  
class LinkedList:  
    """完整的单链表实现"""  
  
    def __init__(self):  
        """初始化链表，使用虚拟头节点简化边界处理"""  
        self.dummy = ListNode(0)  # 虚拟头节点  
        self.size = 0  # 记录链表长度  
  
    # ================== 1. 插入操作 ==================  
    def add_at_head(self, val):  
        """在头部插入节点"""  
        new_node = ListNode(val)  
        new_node.next = self.dummy.next  
        self.dummy.next = new_node  
        self.size += 1  
  
    def add_at_tail(self, val):  
        """在尾部插入节点"""  
        new_node = ListNode(val)  
        curr = self.dummy  
        while curr.next:  
            curr = curr.next  
        curr.next = new_node  
        self.size += 1  
  
    def add_at_index(self, index, val):  
        """在指定位置插入节点"""  
        if index < 0 or index > self.size:  
            return False  
  
        curr = self.dummy  
        for _ in range(index):  
            curr = curr.next  
  
        new_node = ListNode(val)  
        new_node.next = curr.next  
        curr.next = new_node  
        self.size += 1  
        return True  
  
    # ================== 2. 删除操作 ==================  
    def delete_at_head(self):  
        """删除头节点"""  
        if self.is_empty():  
            return False  
        self.dummy.next = self.dummy.next.next  
        self.size -= 1  
        return True  
  
    def delete_at_tail(self):  
        """删除尾节点"""  
        if self.is_empty():  
            return False  
  
        curr = self.dummy  
        while curr.next and curr.next.next:  
            curr = curr.next  
        curr.next = None  
        self.size -= 1  
        return True  
  
    def delete_at_index(self, index):  
        """删除指定位置节点"""  
        if index < 0 or index >= self.size:  
            return False  
  
        curr = self.dummy  
        for _ in range(index):  
            curr = curr.next  
  
        curr.next = curr.next.next  
        self.size -= 1  
        return True  
  
    def remove(self, val):  
        """删除第一个值为 val 的节点"""  
        curr = self.dummy  
        while curr.next:  
            if curr.next.val == val:  
                curr.next = curr.next.next  
                self.size -= 1  
                return True  
            curr = curr.next  
        return False  
  
    # ================== 3. 查询操作 ==================  
    def get(self, index):  
        """获取指定位置节点的值"""  
        if index < 0 or index >= self.size:  
            return -1  
  
        curr = self.dummy.next  
        for _ in range(index):  
            curr = curr.next  
        return curr.val  
  
    def find(self, val):  
        """查找值为 val 的节点索引，不存在返回 -1"""        curr = self.dummy.next  
        index = 0  
        while curr:  
            if curr.val == val:  
                return index  
            curr = curr.next  
            index += 1  
        return -1  
  
    # ================== 4. 工具方法 ==================  
    def get_size(self):  
        """返回链表长度"""  
        return self.size  
  
    def is_empty(self):  
        """判断链表是否为空"""  
        return self.size == 0  
  
    def clear(self):  
        """清空链表"""  
        self.dummy.next = None  
        self.size = 0  
  
    def to_list(self):  
        """转换为 Python 列表（用于调试）"""  
        result = []  
        curr = self.dummy.next  
        while curr:  
            result.append(curr.val)  
            curr = curr.next  
        return result  
  
    def __str__(self):  
        """打印链表"""  
        return " -> ".join(map(str, self.to_list())) + " -> None"  
  
    # ================== 5. 高级算法 ==================  
    def reverse(self):  
        """反转链表"""  
        prev = None  
        curr = self.dummy.next  
  
        while curr:  
            next_node = curr.next  
            curr.next = prev  
            prev = curr  
            curr = next_node  
  
        self.dummy.next = prev  
  
    def has_cycle(self):  
        """判断是否有环（快慢指针）"""  
        slow = fast = self.dummy.next  
        while fast and fast.next:  
            slow = slow.next  
            fast = fast.next.next  
            if slow == fast:  
                return True  
        return False  
  
# ================== 使用示例 ==================if __name__ == "__main__":  
    llist = LinkedList()  
  
    # 插入  
    llist.add_at_head(1)  
    llist.add_at_tail(3)  
    llist.add_at_index(1, 2)  
    print(llist)  # 1 -> 2 -> 3 -> None  
  
    # 查询  
    print(llist.get(1))  # 2  
    print(llist.find(3))  # 2  
  
    # 删除  
    llist.delete_at_index(1)  
    print(llist)  # 1 -> 3 -> None  
  
    # 反转  
    llist.reverse()  
    print(llist)  # 3 -> 1 -> None
