``` python
import sys  
  
  
def solve():  
    n, k = map(int, sys.stdin.readline().split())  
    arr = list(map(int, sys.stdin.readline().split()))  
  
    def check(x):  
        cnt = 0  
  
        for i in range(n):  
            if arr[i] < x:  
                diff = x - arr[i]  
                step = i + 1  
                cnt += (diff + step - 1) // step  
  
            if cnt > k:  
                return  False  
  
        return cnt <= k  
  
    l = 1  
    r = max(arr) + k * n  
    ans = 0  
  
    while l <= r:  
        mid = (l + r) // 2  
  
        if check(mid):  
            ans = mid  
            l = mid + 1  
        else:  
            r = mid - 1  
  
    print(ans)  
  
  
if __name__ == '__main__':  
    solve()
```
