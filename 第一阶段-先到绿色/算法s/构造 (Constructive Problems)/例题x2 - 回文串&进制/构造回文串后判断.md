import sys  
  
def solve():  
  
    A = int(sys.stdin.readline().strip())  
  
    N = int(sys.stdin.readline().strip())  
  
    # 回文串  
    def f(n, base):  
        arr = []  
        while n > 0:  
            arr.append(n % base)  
            n //= base  
        return arr == arr[::-1]  
  
    # 生成回文  
    total_sum = 0  
  
    max_half_len = (len(str(N)) + 1) // 2  
    limit = 10 ** max_half_len  
  
    for i in range(1, limit):  
        s = str(i)  
        rev_s = s[::-1]  
  
        odd_p = int(s + rev_s[1:])  
  
        even_p = int(s + rev_s)  
  
        if odd_p <= N:  
            if f(odd_p, A):  
                total_sum += odd_p  
        else:  
            break  
  
        if even_p <= N:  
            if f(even_p, A):  
                total_sum += even_p  
  
    print(total_sum)  
  
  
if __name__ == '__main__':  
    solve()
