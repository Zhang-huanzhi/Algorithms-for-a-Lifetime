### 暴力代码
```python
import math  
import sys  
  
def main():  
  
    n = int(sys.stdin.readline().strip())  
    s = sys.stdin.readline().strip()  
  
    # 暴力思路  
    ans = [0.0] * (n + 1)  
    for i in range(n):  
        # 中心扩展？看看距离最近的是？  
        cnt = math.inf  
        for j in range(i, -1, -1):  
            if s[j] == '1':  
                cnt = i - j  
                break  
  
        for j in range(i, n):  
            if s[j] == '1':  
                cnt = min(cnt, j - i)  
                break  
  
        ans[i] = cnt  
  
    print(*ans[:n]) 
  
if __name__ == "__main__":  
    main()
```

# 双向扫描
```python
import math  
import sys  
  
def main():  
  
    n = int(sys.stdin.readline().strip())  
    s = sys.stdin.readline().strip()  
  
    # 双向扫描  
    ans = [math.inf] * (n + 1)  
  
    last = -math.inf  
    for i in range(n):  
        if s[i] == '1':  
            last = i  
        ans[i] = i - last  
  
    last = math.inf  
    for i in range(n - 1, -1, -1):  
        if s[i] == '1':  
            last = i  
        ans[i] = min(ans[i], last - i)  
  
    print(*ans[:n])  
  
if __name__ == "__main__":  
    main()
```
