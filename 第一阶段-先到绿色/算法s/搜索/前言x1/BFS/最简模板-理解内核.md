from collections import deque

def bfs(start, g):
    n = len(g)                  # 点编号: 0..n-1
    visited = [False] * n
    q = deque()

    q.append(start)             # 1) 起点入队
    visited[start] = True       # 2) 起点标记已访问

    while q:                    # 3) 队列不空就一直扩展
        u = q.popleft()         # 取出当前层的一个点

        for v in g[u]:          # 扩展它的邻居
            if visited[v]:      # 已访问就跳过
                continue
            visited[v] = True   # 首次到达就标记
            q.append(v)         # 入队，等待后续扩展

    return visited
