import sys
import math
from collections import Counter

def main():

    h, w = map(int, sys.stdin.readline().strip().split())

    arr = []

    for _ in range(h):
        arr.append(list(sys.stdin.readline().strip()))

    up = 0
    while '#' not in arr[up]:
        up += 1

    down = h-1
    while '#' not in arr[down]:
        down -= 1

    # print(up, down)

    left = 0
    state = False
    for j in range(w):
        for i in range(h):
            if arr[i][j] == '#':
                left = j
                state = True
                break
        if state:
            break

    right = 0
    state = False
    for j in range(w-1, -1, -1):
        for i in range(h-1, -1, -1):
            if arr[i][j] == '#':
                right = j
                state = True
                break
        if state:
            break

    # print(left, right)

    # 输出
    for i in range(up, down+1):
        for j in range(left, right+1):
            print(arr[i][j],end = '')
        print(end='\n')

if __name__ == '__main__':
    main()

# Gemini 的
``` python
import sys


def main():
    # 保持与您一致的输入方式
    h, w = map(int, sys.stdin.readline().strip().split())

    arr = []
    rows = []
    cols = []

    for i in range(h):
        line = sys.stdin.readline().strip()
        arr.append(line)
        # 在读取时顺便记录所有 '#' 的坐标
        for j in range(w):
            if line[j] == "#":
                rows.append(i)
                cols.append(j)

    # 确定上下左右边界
    up, down = min(rows), max(rows)
    left, right = min(cols), max(cols)

    # 利用切片输出，避免了内层循环
    for i in range(up, down + 1):
        print(arr[i][left : right + 1])


if __name__ == "__main__":
    main()
```
