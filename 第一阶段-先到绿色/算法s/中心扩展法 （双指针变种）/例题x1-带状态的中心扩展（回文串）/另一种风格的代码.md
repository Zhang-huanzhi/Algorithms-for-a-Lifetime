import sys


def main():
    arr = list(sys.stdin.readline().strip())
    n = len(arr)
    ans = 0

    # 定义中心扩展函数
    def expand(l, r):
        state = False
        while l >= 0 and r < n:
            if arr[l] != arr[r]:
                if state:
                    break
                else:
                    state = True
            l -= 1
            r += 1
        # 使用你发现的黄金公式，自动兼容奇偶数情况
        return (r - l) // 2

    # 1. 累加奇数长度
    for i in range(n):
        ans += expand(i, i)

    # 2. 累加偶数长度
    for i in range(n):
        ans += expand(i, i + 1)

    print(ans)


if __name__ == "__main__":
    main()
