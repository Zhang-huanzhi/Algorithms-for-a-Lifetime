# 暴力
``` python
import math
import sys
from collections import Counter, deque


def main():

    n, m, c = map(int, sys.stdin.readline().strip().split())

    arr = list(map(int, sys.stdin.readline().strip().split()))

    brr = [0] * (m + 1)
    for i in range(n):
        brr[arr[i]] += 1
    # print(brr)

    total_sum = 0
    for i in range(m):
        sum_met = 0
        pos = (i + 1) % m

        while True:
            sum_met += brr[pos]

            if sum_met >= c:

                x_i = sum_met
                break

            pos = (pos + 1) % m

        total_sum += x_i

    print(total_sum)


if __name__ == "__main__":
    main()
```

# 双指针
``` python
import math
import sys
from collections import Counter, deque


def main():

    n, m, c = map(int, sys.stdin.readline().strip().split())

    arr = list(map(int, sys.stdin.readline().strip().split()))

    counts = Counter(arr)
    brr = sorted(counts.keys())
    k = len(brr)

    p = [ counts[b] for b in brr]

    p_extended = p + p

    yrr = [0] * (k + 1)

    right = 0
    current_sum = 0

    for i in range(k):

        while current_sum < c:
            current_sum += p_extended[right]
            right += 1

        yrr[i + 1] = current_sum
        current_sum -= p_extended[i]

    total_sum = 0
    for i in range(1, k):
        len_val = brr[i] - brr[i - 1]
        total_sum += len_val * yrr[i + 1]

    len_last = m + brr[0] - brr[k-1]
    total_sum += len_last * yrr[1]

    print(total_sum)

if __name__ == "__main__":
    main()
```
