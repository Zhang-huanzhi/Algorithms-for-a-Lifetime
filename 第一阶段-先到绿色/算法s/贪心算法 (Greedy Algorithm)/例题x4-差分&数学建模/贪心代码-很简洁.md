``` python
import sys  
  
def solve():  
  
    n, m = map(int, sys.stdin.readline().strip().split())  
  
    arr = list(map(int, sys.stdin.readline().strip().split()))  
  
    diff = [0] * n  
  
    arr.sort()  
  
    for i in range(1, n):  
        diff[i] = arr[i] - arr[i-1]  
  
    diff.sort()  
  
    print(sum(diff[1:n-m+1]))  
  
if __name__ == '__main__':  
    solve()
```
