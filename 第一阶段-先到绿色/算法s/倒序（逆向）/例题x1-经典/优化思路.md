# 暴力
``` python
import math  
import sys  
  
def main():  
  
    n, q = map(int, sys.stdin.readline().strip().split())  
    arr = list(map(int, sys.stdin.readline().strip().split()))  
  
    # 直接写暴力  
    for _ in range(q):  
        x = int(sys.stdin.readline().strip())  
  
        arr.remove(x)  
        arr.append(x)  
  
    print(*arr)  
  
  
if __name__ == "__main__":  
    main()
```

如果靠模拟的话，需要将remove和append的时间复杂度变为logn，或者1



# 时间戳/虚拟坐标法
``` python
import math  
import sys  
  
  
def main():  
    n, q = map(int, sys.stdin.readline().strip().split())  
    arr = list(map(int, sys.stdin.readline().strip().split()))  
  
    brr = [0] * (n + 1)  
    for i, val in enumerate(arr):  
        brr[val] = i  
  
    step = 0  
    for _ in range(q):  
        x = int(sys.stdin.readline().strip())  
  
        brr[x] = n + step  
        step += 1  
  
    ans = sorted(range(1, n + 1), key = lambda x: brr[x])  
  
    print(*ans)  
  
if __name__ == "__main__":  
    main()
```
