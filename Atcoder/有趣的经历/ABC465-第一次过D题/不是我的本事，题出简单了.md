``` python
import sys
import itertools
from collections import deque


def main():

    t = int(sys.stdin.readline().strip())

    for _ in range(t):
        x, y, k = map(int, sys.stdin.readline().strip().split())

        cnt = 0
        # 随便贪心吧
        while True:

            if x == y:
                break

            if x > y:
                x = x // k
                cnt += 1

                if x == y:
                    break

            else:
                y = y // k
                cnt += 1
    
        print(cnt)

if __name__ == "__main__":
    main()
```
