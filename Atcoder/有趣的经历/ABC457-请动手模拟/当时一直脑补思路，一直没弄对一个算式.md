import sys  
  
def solve():  
  
    n, k = map(int, sys.stdin.readline().strip().split())  
  
    arr = []  
    for i in range(n):  
        xrr = list(map(int, sys.stdin.readline().strip().split()))  
        arr.append(xrr[1:])  
  
    brr = list(map(int, sys.stdin.readline().strip().split()))  
  
    crr = []  
    for i in range(n):  
        num = brr[i]  
        crr.append(len(arr[i]) * num)  
  
    # print(crr)  
  
    cnt = 0  
    x = 0  
    y = 0  
  
    for i in range(n):  
        cnt += crr[i]  
        if cnt >= k:  
            x = i  
            y = (k - (cnt - crr[i]) ) % len(arr[x]) - 1  
            break  
  
    # print(x, y)  
    print(arr[x][y])  
  
if __name__ == '__main__':  
    solve()
